# Dependencias del imageWatcher
#
# En Linux Mint las dependencias reales se instalan con apt (no con pip),
# porque PyGObject (gi) y Xlib enlazan con bibliotecas del sistema.
#
#   sudo apt install python3-gi python3-gi-cairo gir1.2-gtk-3.0 \
#                    python3-pil python3-xlib ffmpeg
#
# En Wayland se usan ademas (opcionales, segun el escritorio):
#   xdg-desktop-portal xdg-desktop-portal-gnome/kde/wlr
#   grim slurp wf-recorder
#
# requirements.txt se deja vacio a proposito: no hace falta pip.
